// ============ BOX DRAWING (ASCII) =============
void printLine(int width) {
    int i;
    for (i = 0; i < width; i++) printf("-");
    printf("\n");
}

void printBoxTop(int width) {
    printf("+");
    int i;
    for (i = 0; i < width - 2; i++) printf("-");
    printf("+\n");
}

void printBoxBottom(int width) {
    printf("+");
    int i;
    for (i = 0; i < width - 2; i++) printf("-");
    printf("+\n");
}

void printBoxMid(int width) {
    printf("+");
    int i;
    for (i = 0; i < width - 2; i++) printf("-");
    printf("+\n");
}

void printBoxText(const char *text, int width) {
    int len = strlen(text);
    int padding = width - 4 - len;
    int left = padding / 2;
    int right = padding - left;
    printf("| ");
    int i;
    for (i = 0; i < left; i++) printf(" ");
    printf("%s", text);
    for (i = 0; i < right; i++) printf(" ");
    printf(" |\n");
}
